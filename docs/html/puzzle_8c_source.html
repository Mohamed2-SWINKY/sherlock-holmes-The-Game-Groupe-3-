86void puzzle_handle_event(PuzzleState *ps, SDL_Event *ev) {
87 if (!ps->run) return;
88
89 if (ev->type == SDL_MOUSEBUTTONDOWN) {
90 int mx = ev->button.x, my = ev->button.y;
91 for (int i = 0; i < 3; i++) {
92 if (!ps->pc[i].placed &&
93 mx >= ps->pc[i].pos.x && mx < ps->pc[i].pos.x + ps->pc[i].pos.w &&
94 my >= ps->pc[i].pos.y && my < ps->pc[i].pos.y + ps->pc[i].pos.h) {
95 ps->dIdx = i;
96 ps->ox = mx - ps->pc[i].pos.x;
97 ps->oy = my - ps->pc[i].pos.y;
98 ps->pc[i].drag = 1;
99 break;
100 }
101 }
102 } else if (ev->type == SDL_MOUSEMOTION) {
103 if (ps->dIdx >= 0) {
104 ps->pc[ps->dIdx].pos.x = ev->motion.x - ps->ox;
105 ps->pc[ps->dIdx].pos.y = ev->motion.y - ps->oy;
106 }
107 } else if (ev->type == SDL_MOUSEBUTTONUP) {
108 if (ps->dIdx < 0) return;
109 ps->pc[ps->dIdx].drag = 0;
110 int cx = ps->pc[ps->dIdx].pos.x + ps->pc[ps->dIdx].pos.w / 2;
111 int cy = ps->pc[ps->dIdx].pos.y + ps->pc[ps->dIdx].pos.h / 2;
112 int tx = ps->target.x + ps->target.w / 2;
113 int ty = ps->target.y + ps->target.h / 2;
114 int d = (int)sqrt((double)((cx-tx)*(cx-tx) + (cy-ty)*(cy-ty)));
115
116 if (d < SNAP) {
117 if (ps->pc[ps->dIdx].correct) {
118 ps->pc[ps->dIdx].pos = ps->target;
119 ps->pc[ps->dIdx].placed = 1;
120 ps->result = 1;
121 ps->run = 0;
122 } else {
123 ps->pc[ps->dIdx].pos = ps->pc[ps->dIdx].orig;
124 }
125 }
126 ps->dIdx = -1;
127 }
128}

130void puzzle_update(PuzzleState *ps) {
131 if (!ps->run) {
132 ps->angle += 5.0;
133 if (ps->angle >= 360.0) ps->angle -= 360.0;
134
135 // Pulsing scale effect
136 static float t = 0.0f;
137 t += 0.1f;
138 ps->scale = 1.0f + 0.2f * sinf(t);
139
140 static Uint32 endT = 0;
141 if (endT == 0) endT = SDL_GetTicks();
142 if (SDL_GetTicks() - endT > 3000) { // Increased delay to see animation
143 ps->over = 1;
144 endT = 0;
145 t = 0.0f;
146 }
147 return;
148 }
149
150 if (SDL_GetTicks() - ps->startTime >= TMAX) {
151 ps->result = 0;
152 ps->run = 0;
153 }
154}
