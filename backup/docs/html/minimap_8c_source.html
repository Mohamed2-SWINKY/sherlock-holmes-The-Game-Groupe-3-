67void afficher_minimap(MiniMap m, SDL_Renderer *re) {
68 // apply shake offset
69 SDL_Rect mapRect = m.pos_map;
70 SDL_Rect pointRect = m.pos_point;
71 SDL_Rect point2Rect = m.pos_point2;
72 mapRect.x += m.shakeOffsetX;
73 mapRect.y += m.shakeOffsetY;
74 pointRect.x += m.shakeOffsetX;
75 pointRect.y += m.shakeOffsetY;
76 point2Rect.x += m.shakeOffsetX;
77 point2Rect.y += m.shakeOffsetY;
78
79 // map background
80 if (m.num_level == 1)
81 SDL_RenderCopy(re, m.img_map1, NULL, &mapRect);
82 else
83 SDL_RenderCopy(re, m.img_map2, NULL, &mapRect);
84
85 // P1 — red dot
86 draw_filled_circle(re,
87 pointRect.x + 3, pointRect.y + 3,
88 4, (SDL_Color){255, 0, 0, 255});
89
90 // P2 — blue dot
91 draw_filled_circle(re,
92 point2Rect.x + 3, point2Rect.y + 3,
93 4, (SDL_Color){0, 100, 255, 255});
94
95 // cadre — slightly larger than the map so it frames around it
96 SDL_Rect cadreRect = {
97 mapRect.x - 6, mapRect.y - 6,
98 mapRect.w + 12, mapRect.h + 12
99 };
100 SDL_RenderCopy(re, m.img_cadre, NULL, &cadreRect);
101}

113void MAJ_minimap(MiniMap *m, int x1, int y1, int x2, int y2) {
114 m->joueurX = x1;
115 m->joueurY = y1;
116 m->joueur2X = x2;
117 m->joueur2Y = y2;
118
119 float ratioX = (float)MINIMAP_WORLD_W / (float)m->pos_map.w;
120 float ratioY = (float)MINIMAP_WORLD_H / (float)m->pos_map.h;
121
122 m->pos_point.x = m->pos_map.x + (int)(x1 / ratioX);
123 m->pos_point.y = m->pos_map.y + (int)(y1 / ratioY);
124
125 m->pos_point2.x = m->pos_map.x + (int)(x2 / ratioX);
126 m->pos_point2.y = m->pos_map.y + (int)(y2 / ratioY);
127}

129/* --- 4.3 : Collision Perfect Pixel --- */
130int collision_PP(MiniMap *m, int x, int y) {
131 SDL_Surface *mask = (m->num_level == 1) ? m->img_mask1 : m->img_mask2;
132 if (!mask) return 0;
133
134 Uint32 *pixels = (Uint32 *)mask->pixels;
135 Uint32 color = pixels[y * mask->w + x];
136
137 Uint8 r, g, b;
138 SDL_GetRGB(color, mask->format, &r, &g, &b);
139
140 return (r == 0 && g == 0 && b == 0) ? 1 : 0;
141}

145 return (a.x < b.x + b.w &&

147 a.y < b.y + b.h &&
148 a.y + a.h > b.y);

167void minimap_update_shake(MiniMap *m) {
168 if (m->shakeTimer > 0) {
169 int intensity = m->shakeIntensity * m->shakeTimer / 15;
170 if (intensity < 1) intensity = 1;
171 m->shakeOffsetX = (rand() % (2 * intensity + 1)) - intensity;
172 m->shakeOffsetY = (rand() % (2 * intensity + 1)) - intensity;
173 m->shakeTimer--;
174 } else {
175 m->shakeOffsetX = 0;
176 m->shakeOffsetY = 0;
177 }
178}
