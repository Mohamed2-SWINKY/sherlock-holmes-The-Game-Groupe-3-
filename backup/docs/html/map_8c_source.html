215void update_falling_box(MapData *m, int p1x, int p1y, int p2x, int p2y)
216{
217 FallingBox *fb = &m->fbox;
218 if (fb->state == BOX_BROKEN) return;
219
220 if (fb->state == BOX_IDLE) {
221 int dx1 = p1x - fb->rect.x;
222 int dy1 = p1y - fb->rect.y;
223 int dx2 = p2x - fb->rect.x;
224 int dy2 = p2y - fb->rect.y;
225 float d1 = sqrtf((float)(dx1*dx1 + dy1*dy1));
226 float d2 = sqrtf((float)(dx2*dx2 + dy2*dy2));
227 if (d1 < 130 || d2 < 130)
228 fb->state = BOX_FALLING;
229 return;
230 }
231
232 fb->vel += 0.5f;
233 if (fb->vel > 4.5f) fb->vel = 4.5f;
234 fb->fy += fb->vel;
235 fb->rect.y = (int)fb->fy;
236
237 if (fb->rect.y >= 358) {
238 fb->rect.y = 358;
239 fb->fy = 358.0f;
240 fb->state = BOX_BROKEN;
241 if (m->boxFallSound) {
242 Mix_PlayChannel(-1, m->boxFallSound, 0);
243 }
244 /* reveal the key hidden inside the box (key 0 in level 1) */
245 m->keys1[0].visible = 1;
246 }

277void tb_texte(SDL_Renderer *r, TTF_Font *font,
278 const char *txt, SDL_Color c,
279 int x, int y, int centrerH)
280{
281 if (!font || !txt || !txt[0]) return;
282 SDL_Surface *s = TTF_RenderUTF8_Blended(font, txt, c);
283 if (!s) return;
284 SDL_Texture *t = SDL_CreateTextureFromSurface(r, s);

286 SDL_Rect dst;

290 dst.y = y;

292 SDL_DestroyTexture(t);

339{

344 cl->entrees[j + 1] = cl->entrees[j];
345 j--;
346 }

351int tb_inserer(ClassementTB *cl, const char *nom, int score, int joueur)
352{
353 if (!nom || nom[0] == '\0') return 0;
354 if (cl->nb >= MAX_SCORES_TB &&
355 score <= cl->entrees[MAX_SCORES_TB - 1].score)
356 return 0;
357 int idx = (cl->nb < MAX_SCORES_TB) ? cl->nb : MAX_SCORES_TB - 1;
358 strncpy(cl->entrees[idx].nom, nom, MAX_NOM_TB - 1);
359 cl->entrees[idx].nom[MAX_NOM_TB - 1] = '\0';
360 cl->entrees[idx].score = score;
361 cl->entrees[idx].joueur = joueur;
362 if (cl->nb < MAX_SCORES_TB) cl->nb++;
363 tb_trier(cl);
364 return 1;
365}

367int tb_saisir_nom(GameContext *ctx, char *nomSortie, const char *promptTitle)
368{
369 char buffer[MAX_NOM_TB] = "";
370 int longueur = 0;
371 int valide = 0;
372 int running = 1;
373
374 SDL_StartTextInput();
375
376 Uint32 dernierCligno = SDL_GetTicks();
377 int curseurVisible = 1;
378
379 SDL_Renderer *r = ctx->renderer;
380 TTF_Font *font = ctx->font;
381
382 SDL_Event ev;
383
384 while (running) {
385 while (SDL_PollEvent(&ev)) {
386 if (ev.type == SDL_QUIT) {
387 running = 0; valide = 0;
388 } else if (ev.type == SDL_KEYDOWN) {
389 switch (ev.key.keysym.sym) {
390 case SDLK_ESCAPE:
391 running = 0; valide = 0;
392 break;
393 case SDLK_RETURN:
394 case SDLK_KP_ENTER:
395 if (longueur > 0) { running = 0; valide = 1; }
396 break;
397 case SDLK_BACKSPACE:
398 if (longueur > 0) buffer[--longueur] = '\0';
399 break;
400 default: break;
401 }
402 } else if (ev.type == SDL_TEXTINPUT) {
403 int ajout = (int)strlen(ev.text.text);
404 if (longueur + ajout < MAX_NOM_TB - 1) {
405 strncat(buffer, ev.text.text,
406 (size_t)(MAX_NOM_TB - longueur - 1));
407 longueur += ajout;
408 }
409 } else if (ev.type == SDL_MOUSEBUTTONDOWN &&
410 ev.button.button == SDL_BUTTON_LEFT) {
411 SDL_Rect btnOk = { (WINDOW_WIDTH - 200) / 2, 380, 200, 50 };
412 int mx = ev.button.x, my = ev.button.y;
413 if (mx >= btnOk.x && mx <= btnOk.x + btnOk.w &&
414 my >= btnOk.y && my <= btnOk.y + btnOk.h &&
415 longueur > 0) {
416 running = 0; valide = 1;
417 }
418 }
419 }
420
421 if (SDL_GetTicks() - dernierCligno >= 500) {
422 curseurVisible = !curseurVisible;
423 dernierCligno = SDL_GetTicks();
424 }
425
426 tb_fond(r);
427 tb_texte(r, font, promptTitle, TB_OR, 0, 80, 1);
428 tb_hline(r, 150, WINDOW_WIDTH - 150, 120, TB_OR);
429 tb_texte(r, font, "Votre score sera sauvegarde dans " SCORE_FILE,
430 TB_GRIS, 0, 140, 1);
431
432 SDL_Rect boite = { (WINDOW_WIDTH - 400) / 2, 220, 400, 55 };
433 tb_fillRect(r, boite, (SDL_Color){30, 30, 60, 200});
434 tb_drawRect(r, boite, TB_BLEU);
435
436 char affiche[MAX_NOM_TB + 2];
437 snprintf(affiche, sizeof(affiche), "%s%s",
438 buffer, curseurVisible ? "|" : " ");
439 tb_texte(r, font, affiche, TB_BLEU, boite.x + 14, boite.y + 14, 0);
440
441 tb_texte(r, font, "Entree : valider | Echap : passer",
442 TB_GRIS, 0, 310, 1);
443
444 int mx2, my2;
445 SDL_GetMouseState(&mx2, &my2);
446 SDL_Rect btnOk = { (WINDOW_WIDTH - 200) / 2, 380, 200, 50 };
447 int survol = (mx2 >= btnOk.x && mx2 <= btnOk.x + btnOk.w &&
448 my2 >= btnOk.y && my2 <= btnOk.y + btnOk.h);
449 SDL_Color cBtn = survol ? TB_VERT : (SDL_Color){40, 150, 70, 255};
450 tb_fillRect(r, btnOk, cBtn);
451 tb_drawRect(r, btnOk, TB_BLANC);
452 tb_texte(r, font, "VALIDER", TB_BLANC, btnOk.x + 60, btnOk.y + 13, 0);
453
454 SDL_RenderPresent(r);
455 SDL_Delay(16);
456 }
457
458 SDL_StopTextInput();
459
460 if (valide) {
461 strncpy(nomSortie, buffer, MAX_NOM_TB - 1);
462 nomSortie[MAX_NOM_TB - 1] = '\0';
463 }
464 return valide;
465}
